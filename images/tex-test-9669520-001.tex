\newcommand{\deftheequation}{%
  \ifx\ID\@empty\relax\else
  \@ifundefined{@equation@\thepage @single}%
      {\renewcommand{\theequation}{\ID--\arabic{equation}}}%
      {\renewcommand{\theequation}{\ID}}%
  \fi
}
\newcommand{\@setequationsingle}[1]{%
 \expandafter\xdef\csname @equation@#1@single\endcsname{\@ne}%
}
\newcommand{\refreshequation}{%
  \ifnum\c@equation=\@ne
    \protected@write\@auxout{}%
      {\string\@setequationsingle{\thepage}}%
  \fi
  \setcounter{equation}{\z@}%
}
